            [MOVIE_COMMAND_START]
            [SlashCommand("[REQUEST_MOVIE_TITLE_NAME]", "[REQUEST_MOVIE_TITLE_DESCRIPTION]")]
            [RequireRoles(new ulong[] { [REQUIRED_MOVIE_ROLE_IDS] })]
            [RequireChannels(new ulong[] { [REQUIRED_CHANNEL_IDS] })]
            public async Task MovieByTitle[MOVIE_CATEGORY_ID](InteractionContext ctx, [Option("[REQUEST_MOVIE_TITLE_OPTION_NAME]", "[REQUEST_MOVIE_TITLE_OPTION_DESCRIPTION]")] string movieTitle)
            {
                try
                {
                    var settings = _serviceProvider.Get<DiscordSettingsProvider>().Provide();

                    if (RequestPauseState.IsPaused)
                    {
                        RequestPauseState.LogBlockedAttempt(ctx.User.Username);
                        var pauseReason = RequestPauseState.Reason;
                        var pauseMessage = string.IsNullOrWhiteSpace(pauseReason)
                            ? "Requests are currently paused. Please try again later."
                            : $"Requests are currently paused: {pauseReason}";
                        await ctx.CreateResponseAsync(InteractionResponseType.ChannelMessageWithSource, new DiscordInteractionResponseBuilder().AsEphemeral(true).WithContent(pauseMessage));
                        return;
                    }

                    await ctx.CreateResponseAsync(InteractionResponseType.ChannelMessageWithSource, new DiscordInteractionResponseBuilder().AsEphemeral(settings.AutomaticallyPurgeCommandMessages).WithContent(Language.Current.DiscordCommandSending));

                    var workFlow = _serviceProvider.Get<MovieWorkflowFactory>().CreateRequestingWorkflow(ctx.Interaction, [MOVIE_CATEGORY_ID]);
                    await workFlow.SearchMovieAsync(movieTitle.Trim());
                }
                catch (System.Exception ex)
                {
                    _logger.LogError(ex, $"Error while searching movies with title {movieTitle} : " + ex.Message);
                }
            }

        [SlashCommand("togglerequests", "Pause or resume new movie/tv requests")]
        [RequireChannels(new ulong[] { [REQUIRED_CHANNEL_IDS] })]
        public async Task ToggleRequests(InteractionContext ctx)
        {
            try
            {
                if (!ctx.Member.Permissions.HasPermission(DSharpPlus.Permissions.Administrator))
                {
                    await ctx.CreateResponseAsync(InteractionResponseType.ChannelMessageWithSource, new DiscordInteractionResponseBuilder().AsEphemeral(true).WithContent("Only server admins can use this command."));
                    return;
                }

                bool newState = !RequestPauseState.IsPaused;
                RequestPauseState.SetPaused(newState);

                string message = newState
                    ? "Requests are now paused. Users will see a message when trying to request until this is turned back on."
                    : "Requests are now resumed.";

                await ctx.CreateResponseAsync(InteractionResponseType.ChannelMessageWithSource, new DiscordInteractionResponseBuilder().AsEphemeral(false).WithContent(message));
            }
            catch (System.Exception ex)
            {
                _logger.LogError(ex, "Error while toggling request pause state: " + ex.Message);
            }
        }
